### 二、 关键组件与接口设计

#### 1. 爬虫核心：`frontend/scripts/crawler_smartsheet/headless_har_crawler.mjs`

```javascript
// frontend/scripts/crawler_smartsheet/headless_har_crawler.mjs
import { chromium } from 'playwright';
import path from 'path';

const DOC_URL = 'https://docs.qq.com/smartsheet/DRmRGYkt4dW53VHds';
const HAR_PATH = path.resolve('../../seeds/docs.qq.com.har');

async function run() {
  // 反自动化规避 1: 禁用自动化标记参数
  const browser = await chromium.launch({
    headless: true,
    args: ['--disable-blink-features=AutomationControlled', '--no-sandbox']
  });

  // 反自动化规避 2: 注入真实桌面指纹、视口与语言环境
  const context = await browser.newContext({
    userAgent: 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36',
    viewport: { width: 1920, height: 1080 },
    locale: 'zh-CN',
    recordHar: { path: HAR_PATH, mode: 'full' } // 完整录制捕获含 ZLIB 响应体
  });

  const page = await context.newPage();
  // 反自动化规避 3: 抹除 navigator.webdriver 属性特征
  await page.addInitScript(() => { Object.defineProperty(navigator, 'webdriver', { get: () => undefined }); });

  console.log('[Crawler] 打开文档主页...');
  await page.goto(DOC_URL, { waitUntil: 'networkidle', timeout: 60000 });

  // 逐子表导航等待判定：查找左侧/顶部 Tab 并模拟真实点击触发 dop-api
  const tabs = await page.locator('.smartsheet-tab-item, [role="tab"], .sheet-tab').all();
  for (let i = 0; i < tabs.length; i++) {
    const tabName = (await tabs[i].textContent())?.trim() || `Tab-${i}`;
    console.log(`[Crawler] 正在导航切换到子表: ${tabName}`);
    
    // 监听网络判定：等待目标 dop-api 携带数据的请求完成
    const [response] = await Promise.all([
      page.waitForResponse(res => res.url().includes('dop-api') && res.status() === 200, { timeout: 15000 }).catch(() => null),
      tabs[i].click()
    ]);
    await page.waitForTimeout(1500); // 留存虚拟滚动及响应落地缓冲时间
  }

  // 关闭上下文以刷盘输出完整的 HAR 文件
  await context.close();
  await browser.close();
  console.log(`[Crawler] 抓取完成，HAR 已保存至: ${HAR_PATH}`);
}

run().catch(err => {
  console.error('[Crawler Failed]', err.message);
  process.exit(1); // 触发退出码，供 BAT 脚本感知并降级
});
```

### 四、 验收标准与 Top 3 风险防范

#### 验收标准 (Acceptance Criteria)
1. **HAR 产物完整性**：单跑爬虫脚本，`seeds/docs.qq.com.har` 大小正常（≥ 2MB），包含 9 个子表的 `dop-api` 及 base64 响应正文。
2. **流水线端到端贯通**：运行 `update_and_deploy_har.bat`，自动完成抓取 -> Python 解码去重 -> D1 迁移 -> Worker 构建发布，全过程无阻塞交互。
3. **容灾降级有效性**：模拟断网或阻断无头浏览器时，脚本正确输出 Warning 并回退至 `Downloads` 目录人工 HAR，不发生批处理崩溃中断。

#### Top 3 核心风险与防御策略
1. **反爬滑块验证码拦截**：
   - *风险*：腾讯安全风控升级，弹出滑块验证码导致无头流程超时挂起。
   - *策略*：捕获超时或命中包含验证码的 DOM 时抛出特定错误码退出；提供预留 Cookie 文件注入点（若转入私域/需要鉴权凭证）。
2. **腾讯前端改版致 Tab 选择器失效**：
   - *风险*：DOM 混淆类名变更，导致无法按预期点击子表。
   - *策略*：采用模糊多选机制（同时探测类名、属性与包含特定关键词的文字容器），并在抓取失败时自动截屏保存至 `logs/crawler_error.png` 供排查。
3. **异步请求被截断与 HAR 不完整写入**：
   - *风险*：Tab 切换未完全触发网络包或提前 `context.close()` 导致 ZLIB 数据丢包。
   - *策略*：显式使用 `waitForResponse` 拦截 `dop-api`，并加入网络静默宽限期（`waitForLoadState('networkidle')`），确保 9 个子表全部落地后才关闭上下文。
